### What competition mode is

Read `docs/competition.md` first (flow, roles, scoring rule, Firestore layout, setup), and CLAUDE.md
decision 15. In short: organizer at `/compete` (laptop + camera) → judges join with one QR code per
competition at `/judge?c=<id>&k=<key>` (phone; sign in or guest), each with one role D / A / E →
organizer starts each participant (confirm dialog), the existing AI camera session judges and
**suggests** live to the judges' phones (one tap adds a suggestion to the judge's sheet) → Finish
performance → judges send once → organizer enters penalties and saves → spectators' leaderboard at
`/compete/board?c=<id>` (totals only); organizer sees every judge's score and sheet.
Final = D + A + E − penalties; several judges per role are averaged, with 4+ the highest and lowest
are left out (a TEAM rule, no CoP text in the repo for panel sizes).

// Tiny Chrome DevTools Protocol driver (Node 22 built-in WebSocket). For visual checks only.
import { spawn } from 'node:child_process';
import { writeFileSync } from 'node:fs';

const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

export async function launch({ port, profile, width = 1440, height = 1000, mobile = false }) {
  const proc = spawn('google-chrome', [
    '--headless=new', '--disable-gpu', '--no-sandbox', '--hide-scrollbars', `--remote-debugging-port=${port}`,
    `--user-data-dir=${profile}`, `--window-size=${width},${height}`, '--no-first-run', 'about:blank',
  ], { stdio: 'ignore' });
  let targets;
  for (let i = 0; i < 50; i++) {
    try { targets = await (await fetch(`http://127.0.0.1:${port}/json/list`)).json(); break; } catch { await sleep(200); }
  }
  const page = targets.find((t) => t.type === 'page');
  const ws = new WebSocket(page.webSocketDebuggerUrl);
  await new Promise((r) => ws.addEventListener('open', r, { once: true }));
  let id = 0;
  const waiting = new Map();
  ws.addEventListener('message', (e) => {
    const msg = JSON.parse(e.data);
    if (msg.id && waiting.has(msg.id)) { waiting.get(msg.id)(msg); waiting.delete(msg.id); }
  });
  const send = (method, params = {}) => new Promise((resolve, reject) => {
    const n = ++id;
    waiting.set(n, (msg) => (msg.error ? reject(new Error(msg.error.message)) : resolve(msg.result)));
    ws.send(JSON.stringify({ id: n, method, params }));
  });
  await send('Page.enable');
  await send('Runtime.enable');
  await send('Emulation.setFocusEmulationEnabled', { enabled: true }); // focus / blur work in headless
  if (mobile) await send('Emulation.setDeviceMetricsOverride', { width, height, deviceScaleFactor: 2, mobile: true });
  else await send('Emulation.setDeviceMetricsOverride', { width, height, deviceScaleFactor: 1, mobile: false });

  const evaluate = async (expression) => {
    const r = await send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true });
    if (r.exceptionDetails) throw new Error(r.exceptionDetails.exception?.description ?? 'eval failed');
    return r.result.value;
  };
  return {
    send, evaluate, sleep,
    async goto(url, wait = 2500) { await send('Page.navigate', { url }); await sleep(wait); },
    /** Wait until a JS condition is true (string expression). */
    async waitFor(expr, timeout = 20000) {
      const end = Date.now() + timeout;
      while (Date.now() < end) { if (await evaluate(`Boolean(${expr})`)) return; await sleep(250); }
      throw new Error(`timeout waiting for ${expr}`);
    },
    /** Click the first button/link whose text includes `text`. */
    async click(text) {
      const ok = await evaluate(`(() => { const el = [...document.querySelectorAll('button,a,label')].find(e => e.textContent.trim().includes(${JSON.stringify(text)}) && !e.disabled); if (!el) return false; el.click(); return true; })()`);
      if (!ok) throw new Error(`no clickable "${text}"`);
    },
    /** Type into an input found by CSS selector (React-friendly). */
    async type(selector, value) {
      await evaluate(`(() => { const el = document.querySelector(${JSON.stringify(selector)}); const set = Object.getOwnPropertyDescriptor(Object.getPrototypeOf(el), 'value').set; set.call(el, ${JSON.stringify(value)}); el.dispatchEvent(new Event('input', { bubbles: true })); })()`);
    },
    /** Choose a file in the first <input type=file> matching selector. */
    async setFile(selector, path) {
      const { root } = await send('DOM.getDocument', { depth: -1 });
      const { nodeId } = await send('DOM.querySelector', { nodeId: root.nodeId, selector });
      await send('DOM.setFileInputFiles', { files: [path], nodeId });
    },
    async shot(file, fullPage = true) {
      let clip;
      if (fullPage) {
        const { cssContentSize } = await send('Page.getLayoutMetrics');
        clip = { x: 0, y: 0, width: cssContentSize.width, height: Math.min(cssContentSize.height, 4000), scale: 1 };
      }
      const { data } = await send('Page.captureScreenshot', { format: 'png', ...(clip ? { clip, captureBeyondViewport: true } : {}) });
      writeFileSync(file, Buffer.from(data, 'base64'));
    },
    close() { ws.close(); proc.kill(); },
  };
}

// Three judge "phones" (node): join with the QR key, wait until a participant is being scored,
// send a score for their role, once per participant. Stops after `rounds` participants.
// Usage (from the repo root): npx vite-node path/to/judge-bots.ts <cid> <key> <rounds> [Name1,Name2,Name3]
import { readFileSync } from 'node:fs';
const REPO = process.cwd(); // run from the repo root
for (const line of readFileSync(`${REPO}/.env`, 'utf8').split('\n')) {
  const m = /^(NEXT_PUBLIC_FIREBASE_\w+)=(.*)$/.exec(line.trim());
  if (m) process.env[m[1]] = m[2].trim();
}
const { initializeApp } = await import('firebase/app');
const { getAuth, signInAnonymously } = await import('firebase/auth');
const fs = await import('firebase/firestore');
const { firebaseConfigFromProcessEnv } = await import(`${REPO}/lib/firebase/config.ts`);
const config = (firebaseConfigFromProcessEnv() as { ok: true; config: object }).config;
const [cid, key, roundsText, namesText] = process.argv.slice(2);
const rounds = Number(roundsText ?? 1);
const VALUES: Record<string, number> = { D: 0.2, A: 8.6, E: 9.1 };

await Promise.all((namesText ?? 'Saraa,Bold,Anu').split(',').map(async (name, i) => {
  const app = initializeApp(config, name);
  const auth = getAuth(app);
  await signInAnonymously(auth);
  const uid = auth.currentUser!.uid;
  const db = fs.getFirestore(app);
  await fs.setDoc(fs.doc(db, 'competitions', cid, 'judges', uid), { name, joinKey: key, joinedAt: fs.serverTimestamp(), role: null });
  console.log(`${name} joined`);
  const sentFor = new Set<string>();
  let lastLive = '';
  await new Promise<void>((done) => {
    fs.onSnapshot(fs.doc(db, 'competitions', cid, 'live', 'current'), (snap) => {
      const l = snap.data();
      const k = JSON.stringify(l?.moves?.map((m: { name: string }) => m.name)) + (l?.now?.pose ?? '');
      if (i === 0 && l && k !== lastLive) { lastLive = k; console.log(`  live: moves=[${l.moves.map((m: { name: string; dbValue: number }) => `${m.name} ${m.dbValue}`).join(', ')}] now=${l.now?.pose ?? '-'}`); }
    }, () => undefined);
    fs.onSnapshot(fs.doc(db, 'competitions', cid), async (snap) => {
      const c = snap.data();
      if (!c || c.phase !== 'scoring' || !c.current || sentFor.has(c.current.pid)) return;
      sentFor.add(c.current.pid);
      const me = (await fs.getDoc(fs.doc(db, 'competitions', cid, 'judges', uid))).data()!;
      await new Promise((r) => setTimeout(r, 800 + i * 1200)); // judges send one after another
      const k = Number(c.current.order);
      const value = Math.round((VALUES[me.role] + (me.role === 'D' ? 0.4 * (k % 3) : me.role === 'A' ? -0.3 * ((k + 1) % 3) : -0.2 * (k % 2))) * 100) / 100;
      await fs.setDoc(fs.doc(db, 'competitions', cid, 'participants', c.current.pid, 'scores', uid), { role: me.role, value, items: [], submittedAt: fs.serverTimestamp() });
      console.log(`${name} (${me.role}) sent ${value} for ${c.current.name}`);
      if (sentFor.size >= rounds) done();
    }, (e) => console.log(name, 'error', e.message));
  });
}));
console.log('bots done');
process.exit(0);
